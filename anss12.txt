using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;

namespace LINQ_Assignment_01
{
    // Custom Comparer for Case-Insensitive Sorting
    public class CaseInsensitiveComparer : IComparer<string>
    {
        public int Compare(string x, string y)
        {
            return string.Compare(x, y, StringComparison.OrdinalIgnoreCase);
        }
    }

    class Program
    {
        static void Main(string[] args)
        {
            Console.WriteLine("==================================================");
            Console.WriteLine("             LINQ ASSIGNMENT 01 SOLUTIONS         ");
            Console.WriteLine("==================================================\n");

            #region 1. LINQ - Restriction Operators

            Console.WriteLine("--- 1. LINQ - Restriction Operators ---");

            // 1. Find all products that are out of stock.
            Console.WriteLine("\n[Q1] Out of stock products:");
            var outOfStock = ListGenerators.ProductList.Where(p => p.UnitsInStock == 0);
            foreach (var p in outOfStock)
                Console.WriteLine($" - {p.ProductName}");

            // 2. Find all products that are in stock and cost more than 3.00 per unit.
            Console.WriteLine("\n[Q2] In stock and Price > 3.00:");
            var inStockPrice3 = ListGenerators.ProductList.Where(p => p.UnitsInStock > 0 && p.UnitPrice > 3.00m);
            foreach (var p in inStockPrice3.Take(5)) // Taking 5 for display
                Console.WriteLine($" - {p.ProductName}: {p.UnitPrice:C}");

            // 3. Returns digits whose name is shorter than their value.
            Console.WriteLine("\n[Q3] Digits whose name is shorter than their value:");
            string[] digitsArr = { "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine" };
            var shortDigits = digitsArr.Where((name, index) => name.Length < index);
            foreach (var d in shortDigits)
                Console.WriteLine($" - {d}");

            #endregion

            #region 2. LINQ - Element Operators

            Console.WriteLine("\n--------------------------------------------------");
            Console.WriteLine("--- 2. LINQ - Element Operators ---");

            // 1. Get first Product out of Stock
            Console.WriteLine("\n[Q1] First Product out of Stock:");
            var firstOutOfStock = ListGenerators.ProductList.FirstOrDefault(p => p.UnitsInStock == 0);
            Console.WriteLine($" - {firstOutOfStock?.ProductName ?? "None"}");

            // 2. Return the first product whose Price > 1000, unless there is no match (null returned)
            Console.WriteLine("\n[Q2] First product with Price > 1000:");
            var priceOver1000 = ListGenerators.ProductList.FirstOrDefault(p => p.UnitPrice > 1000);
            Console.WriteLine($" - {(priceOver1000 != null ? priceOver1000.ProductName : "null")}");

            // 3. Retrieve the second number greater than 5
            Console.WriteLine("\n[Q3] Second number greater than 5:");
            int[] numbersArr = { 5, 4, 1, 3, 9, 8, 6, 7, 2, 0 };
            var secondGreaterThan5 = numbersArr.Where(n => n > 5).ElementAtOrDefault(1);
            Console.WriteLine($" - {secondGreaterThan5}");

            #endregion

            #region 3. LINQ - Aggregate Operators

            Console.WriteLine("\n--------------------------------------------------");
            Console.WriteLine("--- 3. LINQ - Aggregate Operators ---");

            // 1. Uses Count to get the number of odd numbers in the array
            Console.WriteLine("\n[Q1] Number of odd numbers:");
            int oddCount = numbersArr.Count(n => n % 2 != 0);
            Console.WriteLine($" - {oddCount}");

            // 2. Return a list of customers and how many orders each has.
            Console.WriteLine("\n[Q2] Customers and order count:");
            var customerOrderCounts = ListGenerators.CustomerList.Select(c => new { c.CustomerName, OrderCount = c.Orders.Length });
            foreach (var c in customerOrderCounts.Take(5))
                Console.WriteLine($" - {c.CustomerName}: {c.OrderCount} orders");

            // 3. Return a list of categories and how many products each has.
            Console.WriteLine("\n[Q3] Categories and product count:");
            var categoryCounts = ListGenerators.ProductList.GroupBy(p => p.Category).Select(g => new { Category = g.Key, ProductCount = g.Count() });
            foreach (var c in categoryCounts)
                Console.WriteLine($" - {c.Category}: {c.ProductCount} products");

            // 4. Get the total of the numbers in an array.
            Console.WriteLine("\n[Q4] Total sum of array:");
            int totalSum = numbersArr.Sum();
            Console.WriteLine($" - Sum: {totalSum}");

            // 5-8. Dictionary Statistics
            string dictPath = "dictionary_english.txt";
            if (File.Exists(dictPath))
            {
                string[] wordsDict = File.ReadAllLines(dictPath);

                Console.WriteLine($"\n[Q5] Total characters in dictionary: {wordsDict.Sum(w => (long)w.Length)}");
                Console.WriteLine($"[Q6] Shortest word length: {wordsDict.Min(w => w.Length)}");
                Console.WriteLine($"[Q7] Longest word length: {wordsDict.Max(w => w.Length)}");
                Console.WriteLine($"[Q8] Average word length: {wordsDict.Average(w => w.Length):F2}");
            }
            else
            {
                Console.WriteLine("\n[Q5-Q8] (dictionary_english.txt not found in bin directory)");
            }

            #endregion

            #region 4. LINQ - Ordering Operators

            Console.WriteLine("\n--------------------------------------------------");
            Console.WriteLine("--- 4. LINQ - Ordering Operators ---");

            // 1. Sort a list of products by name
            Console.WriteLine("\n[Q1] Products sorted by name:");
            var sortedByName = ListGenerators.ProductList.OrderBy(p => p.ProductName);
            foreach (var p in sortedByName.Take(3))
                Console.WriteLine($" - {p.ProductName}");

            // 2. Custom comparer case-insensitive sort
            Console.WriteLine("\n[Q2] Case-insensitive sort:");
            string[] wordsArr = { "aPPLE", "AbAcUs", "bRaNcH", "BlUeBeRrY", "ClOvEr", "cHeRry" };
            var customSorted = wordsArr.OrderBy(w => w, new CaseInsensitiveComparer());
            foreach (var w in customSorted)
                Console.WriteLine($" - {w}");

            // 3. Sort by units in stock from highest to lowest
            Console.WriteLine("\n[Q3] Products by stock (High to Low):");
            var sortedByStock = ListGenerators.ProductList.OrderByDescending(p => p.UnitsInStock);
            foreach (var p in sortedByStock.Take(3))
                Console.WriteLine($" - {p.ProductName}: {p.UnitsInStock}");

            // 4. Sort digits by length, then alphabetically
            Console.WriteLine("\n[Q4] Digits sorted by length then name:");
            var sortedDigits = digitsArr.OrderBy(d => d.Length).ThenBy(d => d);
            foreach (var d in sortedDigits)
                Console.WriteLine($" - {d}");

            // 5. Sort by length then case-insensitive
            Console.WriteLine("\n[Q5] Sort by length then case-insensitive:");
            var sortedLenCase = wordsArr.OrderBy(w => w.Length).ThenBy(w => w, StringComparer.OrdinalIgnoreCase);
            foreach (var w in sortedLenCase)
                Console.WriteLine($" - {w}");

            // 6. Sort products by category then unit price descending
            Console.WriteLine("\n[Q6] Products by Category then Price Descending:");
            var sortedCatPrice = ListGenerators.ProductList.OrderBy(p => p.Category).ThenByDescending(p => p.UnitPrice);
            foreach (var p in sortedCatPrice.Take(4))
                Console.WriteLine($" - [{p.Category}] {p.ProductName}: {p.UnitPrice:C}");

            // 7. Sort by length then case-insensitive descending
            Console.WriteLine("\n[Q7] Sort by length then case-insensitive descending:");
            var sortedLenCaseDesc = wordsArr.OrderBy(w => w.Length).ThenByDescending(w => w, StringComparer.OrdinalIgnoreCase);
            foreach (var w in sortedLenCaseDesc)
                Console.WriteLine($" - {w}");

            // 8. Digits with 2nd letter 'i' reversed
            Console.WriteLine("\n[Q8] Digits with 2nd letter 'i' reversed:");
            var digitsIReversed = digitsArr.Where(d => d.Length > 1 && d[1] == 'i').Reverse();
            foreach (var d in digitsIReversed)
                Console.WriteLine($" - {d}");

            #endregion

            #region 5. LINQ - Transformation Operators

            Console.WriteLine("\n--------------------------------------------------");
            Console.WriteLine("--- 5. LINQ - Transformation Operators ---");

            // 1. Sequence of product names
            Console.WriteLine("\n[Q1] Product Names sequence:");
            var productNames = ListGenerators.ProductList.Select(p => p.ProductName);
            foreach (var name in productNames.Take(3))
                Console.WriteLine($" - {name}");

            // 2. Uppercase and Lowercase versions (Anonymous Types)
            Console.WriteLine("\n[Q2] Upper and Lowercase versions:");
            string[] sampleWords = { "aPPLE", "BlUeBeRrY", "cHeRry" };
            var upperLower = sampleWords.Select(w => new { Upper = w.ToUpper(), Lower = w.ToLower() });
            foreach (var item in upperLower)
                Console.WriteLine($" - Upper: {item.Upper}, Lower: {item.Lower}");

            // 3. Rename UnitPrice to Price
            Console.WriteLine("\n[Q3] Renamed property UnitPrice to Price:");
            var renamedProp = ListGenerators.ProductList.Select(p => new { p.ProductName, Price = p.UnitPrice });
            foreach (var item in renamedProp.Take(3))
                Console.WriteLine($" - {item.ProductName}: {item.Price:C}");

            // 4. Number In-place matching
            Console.WriteLine("\n[Q4] Number: In-place?");
            Console.WriteLine("Number: In-place?");
            var inPlaceResult = numbersArr.Select((num, index) => new { Number = num, InPlace = (num == index) });
            foreach (var item in inPlaceResult)
                Console.WriteLine($"{item.Number}: {item.InPlace}");

            // 5. Pairs where numbersA < numbersB
            Console.WriteLine("\n[Q5] Pairs where a < b:");
            int[] numbersA = { 0, 2, 4, 5, 6, 8, 9 };
            int[] numbersB = { 1, 3, 5, 7, 8 };

            Console.WriteLine("Pairs where a < b:");
            var pairs = from a in numbersA
                        from b in numbersB
                        where a < b
                        select new { a, b };

            foreach (var pair in pairs)
                Console.WriteLine($"{pair.a} is less than {pair.b}");

            // 6. Orders where total < 500.00
            Console.WriteLine("\n[Q6] Orders with total < 500.00:");
            var ordersLessThan500 = ListGenerators.CustomerList
                .SelectMany(c => c.Orders)
                .Where(o => o.Total < 500.00m);
            foreach (var o in ordersLessThan500.Take(3))
                Console.WriteLine($" - OrderID: {o.OrderID}, Total: {o.Total:C}");

            // 7. Orders made in 1998 or later
            Console.WriteLine("\n[Q7] Orders made in 1998 or later:");
            var orders1998OrLater = ListGenerators.CustomerList
                .SelectMany(c => c.Orders)
                .Where(o => o.OrderDate.Year >= 1998);
            foreach (var o in orders1998OrLater.Take(3))
                Console.WriteLine($" - OrderID: {o.OrderID}, Date: {o.OrderDate:yyyy-MM-dd}");

            #endregion

            Console.WriteLine("\n==================================================");
            Console.WriteLine("                END OF ASSIGNMENT                ");
            Console.WriteLine("==================================================");
        }
    }
}
